## 二、执行脚本（映射表为待填配置——唯一的开放变量）

```python
# S2_lambda_closure_v1.py — 预注册 S2-REG-20260918-01
# 用法: 填好 MAPPING 后运行; 判据与样本池已冻结, 改判据=作废重注册
import numpy as np, json

TC_TRAIN = {"Nb":9.3,"Pb":7.2,"Nb3Sn":18.3,"MgB2":39,"YBCO":92,
            "Bi-2212":85,"LaFeAsO-F":26,"FeSe":8,"H3S_hp":203,"LaH10_hp":250}
TC_HOLD  = {"NbN":16,"CaKFe4As4":35,"Cs3C60":38}

# 【开放变量·待知世选定】材料→扇区映射表(每材料一个N或"未映射")
# 候选映射假设(由知世/体系给, ZCode 不代定):
#   M-A: 全部常规/合金超导→N=8, 铜基→N=12, 铁基→N=16, 氢化物→N=24
#   M-B: 按 Tc 量级分档映射
#   M-C: 知世自定
MAPPING = {}   # 例: {"Nb":8, "YBCO":12, ...}  填好后运行

# 扇区特征量(统一维度表直算, [体系]口径; 未映射材料自动剔除并计数)
def sector_feats(N):
    v2 = 0
    while N % 2 == 0 and N > 1:
        N //= 2; v2 += 1
    alpha = 2.0**(-v2) if N == 1 else None   # 纯2幂支口径; 非纯2幂支按生成公式另行扩展
    return {"alpha": alpha}

def model_sym(alpha_vec, coef):
    # Λ-① 候选函数族: Tc_pred = c0 + c1*alpha + c2*alpha^2 (最小二乘)
    A = np.column_stack([np.ones(len(alpha_vec)), alpha_vec, np.array(alpha_vec)**2])
    c, *_ = np.linalg.lstsq(A, coef, rcond=None)
    return A @ c, c

def run():
    assert MAPPING, "映射表未填——按预注册, 映射由知世选定"
    unmapped = [k for k in TC_TRAIN if k not in MAPPING]
    mats = [k for k in TC_TRAIN if k in MAPPING]
    feats = [sector_feats(MAPPING[k])["alpha"] for k in mats]
    y = np.array([TC_TRAIN[k] for k in mats])
    pred, coef = model_sym(feats, y)
    r2 = 1 - ((y-pred)**2).sum()/((y-y.mean())**2).sum()
    verdict = "PASS" if r2 >= 0.6 else "判负(训练R²<0.6)"
    out = {"预注册":"S2-REG-20260918-01","映射":MAPPING,"未映射剔除":unmapped,
           "训练R2":round(float(r2),4),"系数":coef.tolist(),
           "判负规则":"R²<0.6判负; held-out≥2项>20%判负; 三路径全负复活关闭",
           "VERDICT_train":verdict,"held_out":"封存至训练冻结+映射选定"}
    print(json.dumps(out, ensure_ascii=False, indent=1))
    with open("S2_lambda_result.json","w",encoding="utf-8") as f:
        json.dump(out,f,ensure_ascii=False,indent=1)

if __name__ == "__main__":
    run()
```
